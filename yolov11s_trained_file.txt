from ultralytics import YOLO

model = YOLO('yolo11s.pt')

model.train(
    data='/content/YOLO_Dataset/data.yaml',
    epochs=100,
    imgsz=640,
    batch=16,
    device=0,
    workers=8,
    patience=50,
    lr0=0.001,
    optimizer='AdamW',
    project='runs/vehicle_detection',
    name='yolo11s_vehicle',
    exist_ok=True
)

print("✅ Training Complete!")

from ultralytics import YOLO
import cv2
from google.colab.patches import cv2_imshow
import os

# Load model
model = YOLO('/content/runs/detect/runs/vehicle_detection/yolo11s_vehicle/weights/best.pt')

# Test on validation images
val_dir = '/content/YOLO_Dataset/images/val/'
val_images = os.listdir(val_dir)[:5]

for img_name in val_images:
    img_path = os.path.join(val_dir, img_name)
    results = model(img_path)

    print(f"\n📷 {img_name}: {len(results[0].boxes)} vehicles detected")
    cv2_imshow(results[0].plot())
