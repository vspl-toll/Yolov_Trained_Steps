# ============================================================
# STEP 6: YOLOv5s TRAINING
# ============================================================

from ultralytics import YOLO

print("\n" + "=" * 60)
print("🚀 YOLOv5s TRAINING STARTING")
print("=" * 60)

# YOLOv5 Small pretrained model
model = YOLO('yolov5s.pt')

print("\n✅ YOLOv5s model loaded!")

# Training
results = model.train(
    data='/content/YOLO_Dataset/data.yaml',

    # Training settings
    epochs=100,
    imgsz=640,
    batch=16,

    # GPU
    device=0,

    # Workers
    workers=8,

    # Early stopping
    patience=50,

    # Learning rate
    lr0=0.001,

    # Optimizer
    optimizer='AdamW',

    # Output folder
    project='/content/runs/vehicle_detection',

    # Experiment name
    name='yolov5s_vehicle',

    # Existing folder overwrite
    exist_ok=True
)

print("\n" + "=" * 60)
print("✅ YOLOv5s TRAINING COMPLETE!")
print("=" * 60)

# STEP 10: Validate on Validation Images
# ============================================================

print("\n" + "=" * 60)
print("🧪 VALIDATION IMAGE TEST")
print("=" * 60)

from ultralytics import YOLO
import cv2
from google.colab.patches import cv2_imshow

# Load trained YOLOv5s model
if os.path.exists(BEST_MODEL):

    model = YOLO(BEST_MODEL)

    print("\n✅ Trained YOLOv5s model loaded!")

    # Validation image folder
    val_dir = '/content/YOLO_Dataset/images/val/'

    # Get image files
    val_images = [
        f for f in os.listdir(val_dir)
        if f.lower().endswith(('.jpg', '.jpeg', '.png', '.bmp'))
    ]

    # First 5 images
    val_images = val_images[:5]

    print("\n📷 Testing", len(val_images), "validation images...")

    for img_name in val_images:

        img_path = os.path.join(val_dir, img_name)

        # Inference
        results = model(img_path)

        # Number of detections
        detection_count = len(results[0].boxes)

        print("\n" + "-" * 50)
        print("📷 Image:", img_name)
        print("🚗 Vehicles detected:", detection_count)

        # Show detection result
        annotated_image = results[0].plot()

        cv2_imshow(annotated_image)

else:

    print("\n❌ Trained best.pt not found!")
    print("Validation skipped.")
